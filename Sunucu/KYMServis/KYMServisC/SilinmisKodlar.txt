         public KullaniciVAN KullaniciOkuSession(string sessionId)
        {
            try
            {
                return new KullaniciServis().OkuSession(null, sessionId);
            }
            catch (Exception e)
            {
                KullaniciVAN donusDeger = new KullaniciVAN();
                donusDeger.sonuc.islemSonuc = false;
                donusDeger.sonuc.hataStr = e.ToString();
                return donusDeger;
            }
        }

        public Sonuc KullaniciSessionSil(string sessionId)
        {
            try
            {
                return new KullaniciServis().SessionSil(null, sessionId);
            }
            catch (Exception e)
            {
                Sonuc sonuc = new Sonuc();
                sonuc.islemSonuc = false;
                sonuc.hataStr = e.ToString();
                return sonuc;
            }
        }
