        private static Sonuc MasrafPlaniGenelOtomatikHesaplaIc(Kullanici kullanan, VeriErisim veriErisim, int yil, KurumsalYapi kurumsalYapi, string ekoKod2, double artisYuzde, bool ustuneYaz, bool bosIseYaz)

                }

                //Sonra dağıt;
                if (gelenToplam > 0)
                {
                    double dagitToplam = 0;
                    for (int i = 0; i < to.Length; i++)
                    {
                        for (int j = 0; j < 12; j++)
                        {
                            if ((i == to.Length - 1) && (j == 11))
                            {
                                to[i].odenek[j] = olacakToplam - dagitToplam;
                            }
                            else
                            {
                                to[i].odenek[j] = Math.Floor(to[i].odenek[j] * (olacakToplam / gelenToplam));
                                dagitToplam += to[i].odenek[j];
                            }
                        }
                    }
                    return PerformansTertibeCevir(kullanan, yil, to, false);
                }
                else
                    return null;
            }
            else
                return null;
        }

        private static TertipOdenek[] PerformansTertibeCevir(Kullanici kullanan, int yil, TertipOdenek[] to, bool hataVer)
        {
            ArrayList al = new ArrayList();
            for (int i = 0; i < to.Length; i++)
            {
                TertipOdenek[] toIc = PEBTertipleriBul(kullanan, to[i].tertip);
                if (toIc != null && toIc.Length > 0)
                {
                    double odenekTop = 0;
                    for (int j = 0; j < toIc.Length; j++)
                    {
                        odenekTop += toIc[j].odenek[0];
                        toIc[j].odenek = new double[to[i].odenek.Length];
                        al.Add(toIc[j]);
                    }

                    for (int ayIc = 0; ayIc < to[i].odenek.Length; ayIc++)
                    {

                        double dagitTerTop = to[i].odenek[ayIc];
                        double dagitilanTerTop = 0;
                        for (int j = 0; j < toIc.Length; j++)
                        {
                            double oran = 0;
                            if (odenekTop > 0)
                                oran = toIc[j].odenek[0] / odenekTop;
                            else
                                oran = 1;
                            if ((j == toIc.Length - 1))
                            {
                                toIc[j].odenek[ayIc] = dagitTerTop - dagitilanTerTop;
                            }
                            else
                            {
                                toIc[j].odenek[ayIc] = Math.Floor(to[i].odenek[ayIc] * oran);
                                dagitilanTerTop += toIc[j].odenek[ayIc];
                            }
                        }
                    }
                }
                else if (hataVer)
                {
                    throw new Exception(to[i].tertip.TertipStrOl() + " Tertibine ait performans bilgileri bulunamadı!");
                }
            }
            return (TertipOdenek[])al.ToArray(typeof(TertipOdenek));
        }

        private static TertipOdenek[] AylikDagilimdanTertipOdenege(int yil, ObjectArray oa, bool artimli, bool tahminli, int artimAy)
        {
            ArrayList al = new ArrayList();
            for (int i = 0; i < oa.ObjeSayisi; i++)
            {
                TNS.MUH.AylikDagilim ad = (TNS.MUH.AylikDagilim)oa[i];

                if (!ad.tertip.TamMi())
                    throw new Exception(ad.tertip.TertipStrOl() + " hatalı tertip geldi");

                TertipOdenek aktifTo = ArrayListteTertipBul(al, ad.tertip);
                if (aktifTo == null)
                {
                    aktifTo = new TertipOdenek(12);
                    aktifTo.tertip = ad.tertip;
                    aktifTo.tertip.yil = yil;
                    al.Add(aktifTo);
                }
                aktifTo.odenek[ad.ay - 1] = ad.tutar;
            }
            double oncekiAyDeger = 0;
            if (artimli)
            {
                for (int i = 0; i < al.Count; i++)
                {
                    TertipOdenek aktifTo = (TertipOdenek)al[i];
                    oncekiAyDeger = aktifTo.odenek[0];
                    for (int ayIc = 1; ayIc < artimAy; ayIc++)
                    {
                        double simdikiAyDeger = aktifTo.odenek[ayIc];
                        aktifTo.odenek[ayIc] = simdikiAyDeger - oncekiAyDeger;
                        oncekiAyDeger = simdikiAyDeger;
                    }
                    if (oncekiAyDeger == 0)
                    {
                        al.RemoveAt(i);
                        i--;
                    }
                    else
                    {
                        for (int ayIc = artimAy; ayIc < 12; ayIc++)
                        {
                            aktifTo.odenek[ayIc] = 0;
                        }
                        for (int ayIc = 0; ayIc < 12; ayIc++)
                        {
                            if (aktifTo.odenek[ayIc] < 0)
                            {
                                for (int dusurAy = ayIc - 1; dusurAy >= 0 && aktifTo.odenek[ayIc] < 0; dusurAy--)
                                {
                                    if (aktifTo.odenek[dusurAy] + aktifTo.odenek[ayIc] >= 0)
                                    {
                                        aktifTo.odenek[dusurAy] = aktifTo.odenek[dusurAy] + aktifTo.odenek[ayIc];
                                        aktifTo.odenek[ayIc] = 0;
                                    }
                                    else
                                    {
                                        aktifTo.odenek[ayIc] = aktifTo.odenek[dusurAy] + aktifTo.odenek[ayIc];
                                        aktifTo.odenek[dusurAy] = 0;
                                    }
                                }
                                aktifTo.odenek[ayIc] = 0;
                            }
                        }

                    }
                }
            }

            if (tahminli)
            {
                for (int i = 0; i < al.Count; i++)
                {
                    TertipOdenek aktifTo = (TertipOdenek)al[i];
                    oncekiAyDeger = 0;
                    for (int ayIc = 0; ayIc < artimAy; ayIc++)
                    {
                        oncekiAyDeger += aktifTo.odenek[ayIc];
                    }
                    double tut = Math.Floor(oncekiAyDeger / artimAy);
                    for (int ayIc = artimAy; ayIc < 12; ayIc++)
                    {
                        aktifTo.odenek[ayIc] = tut;
                    }
                }
            }
            return (TertipOdenek[])al.ToArray(typeof(TertipOdenek));
        }

        private static TertipOdenek ArrayListteTertipBul(ArrayList al, Tertip tertip)

                if (aktifTo.tertip.TertipStrOl() == tertip.TertipStrOl())
                    return aktifTo;
            }
            return null;
        }

                            }
                        }
                        al.Add(aktifTO);
                    }
                }
            }

        public static Sonuc MasrafPlaniGenelKopyala(VeriErisim veriErisim, Kullanici kullanan, int kaynakYil, ENUMMasrafPlanAsama kaynakAsama, int hedefYil, ENUMMasrafPlanAsama hedefAsama, string kurumsalYapiKod, string kurKod, string ekoKod)
        {
            Sonuc s;
            if (!kullanan.KullaniciTipiIceriyorMu((int)ENUMKullaniciTipi.SISTEMYONETICISI))
                return new Sonuc(false, "Sistem yöneticisi olmalısınız!", "");

            VeriErisim ve = VeriErisim.VeriErisimBul(veriErisim, VERIERISIMAD, kullanan.KimlikBul());
            TABMPGMASRAFPLANI tab = new TABMPGMASRAFPLANI(ve);
            if (string.IsNullOrEmpty(kurumsalYapiKod) || string.IsNullOrEmpty(kurKod) || string.IsNullOrEmpty(ekoKod))
            {
                int kayitSay = tab.KayitSayisiBul(hedefYil, hedefAsama);
                if (kayitSay > 0)
                {
                    ve.VeriErisimKapat();
                    return new Sonuc(false, "Hedef aşamada veri olduğu için kopyalama yapamazsınız!", "");
                }
            }
            ve.BeginTransaction();
            if (!string.IsNullOrEmpty(kurumsalYapiKod) && !string.IsNullOrEmpty(kurKod) && !string.IsNullOrEmpty(ekoKod))
            {
                s = tab.BirimAsamaSil(hedefYil, hedefAsama, kurumsalYapiKod, kurKod, ekoKod);
            }
            s = tab.AsamaKopyala(kaynakYil, kaynakAsama, hedefYil, hedefAsama, kurumsalYapiKod, kurKod, ekoKod);
            if (s.islemSonuc)
                ve.Commit();
            else
                ve.Rollback();
            ve.VeriErisimKapat();
            return s;
        }

        private static double[] HarcamalariBul(Kullanici kullanan, int yil, KurumsalYapi ky, string kurKod, string ekoKod)
        {
            double[] donusDeger = new double[12];
            double oncekiAy = 0;
            for (int ayIc = 0; ayIc < 12; ayIc++)
            {
                double simdikiAy = servisUZY.UzayDegeriDbl(kullanan, "ISLHARCAMAMUHASEBE", UzayKosulOl((yil).ToString(), (ayIc + 1).ToString(), Arac.SaymanlikKodUzayKosulOl(ky.saymanlikKod, true), ky.birimKod, kurKod, "", "", ekoKod), false);
                donusDeger[ayIc] = simdikiAy - oncekiAy;
                oncekiAy = simdikiAy;
            }

            for (int ayIc = 0; ayIc < 12; ayIc++)
            {
                if (donusDeger[ayIc] < 0)
                {
                    for (int dusurAy = ayIc - 1; dusurAy >= 0 && donusDeger[ayIc] < 0; dusurAy--)
                    {
                        if (donusDeger[dusurAy] + donusDeger[ayIc] >= 0)
                        {
                            donusDeger[dusurAy] = donusDeger[dusurAy] + donusDeger[ayIc];
                            donusDeger[ayIc] = 0;
                        }
                        else
                        {
                            donusDeger[ayIc] = donusDeger[dusurAy] + donusDeger[ayIc];
                            donusDeger[dusurAy] = 0;
                        }
                    }
                    donusDeger[ayIc] = 0;
                }
            }
            return donusDeger;
        }

        private static double[] DonemiAyaDagit(double[] ayDagilim, double[] donemler)
        {
            double[] aylar = new double[12];
            for (int i = 0; i < 4; i++)
            {
                double donemToplam = ayDagilim[i * 3 + 0] + ayDagilim[i * 3 + 1] + ayDagilim[i * 3 + 2];
                if (donemToplam != 0)
                {
                    aylar[i * 3 + 0] = Math.Floor(donemler[i] * ayDagilim[i * 3 + 0] / donemToplam);
                    aylar[i * 3 + 1] = Math.Floor(donemler[i] * ayDagilim[i * 3 + 1] / donemToplam);
                }
                else
                {
                    aylar[i * 3 + 0] = Math.Floor(donemler[i] / 3);
                    aylar[i * 3 + 1] = aylar[i * 3 + 0];
                }
                aylar[i * 3 + 2] = donemler[i] - (aylar[i * 3 + 0] + aylar[i * 3 + 1]);
            }
            return aylar;
        }

        public static Sonuc MasrafPlaniMerkezDagitim(VeriErisim veriErisim, Kullanici kullanan, int yil, string kurumsalYapiKod, ENUMMasrafPlanAsama asama, string kurKod, string ekoKod, string[] ekoKodlar, double[] donemler)
        {
            KurumsalYapi ky = BHMv2.IsKurallari.BHMv2KurumsalYapi.KurumsalYapiDondur(veriErisim, kurumsalYapiKod);
            if (ky == null)
                throw new Exception("Kurumsal Yapı Çözümlenemedi");
            if (!YazmaYetkisiVarMi(kullanan, yil, ky, asama, kurKod))
                throw new Exception("Birime yazma yetkiniz yok!");
            if (string.IsNullOrEmpty(kurumsalYapiKod))
                throw new Exception("Bir kurumsal birim seçmelisiniz!");

            for (int i = 0; i < donemler.Length; i++)
            {
                if (donemler[i] < 0)
                    throw new Exception((i + 1).ToString() + ". dönemin değeri sıfırdan küçük (" + donemler[i].ToString("#,###") + ")");
            }
            double[] ayDagilim = HarcamalariBul(kullanan, yil - 1, ky, kurKod, ekoKod);
            ayDagilim = DonemiAyaDagit(ayDagilim, donemler);

            //string tut = "";
            //for (int i = 0; i < 12; i++)
            //{
            //    tut += i.ToString() + ":" + ayDagilim[i].ToString("#,##0.00") + "; ";
            //}
            VeriErisim ve = VeriErisim.VeriErisimBul(veriErisim, VERIERISIMAD, kullanan.KimlikBul());
            double[] eldekiAyDagilim = new double[12];
            for (int i = 0; i < 12; i++)
            {
                eldekiAyDagilim[i] = 0;
            }
            string hataStr = "";
            TABMPGMASRAFPLANI tab = new TABMPGMASRAFPLANI(ve);
            IDataReader dr = tab.AylikDagilimBul(yil, kurumsalYapiKod, asama, kurKod, ekoKod, ekoKodlar);
            if (dr != null)
            {
                while (dr.Read())
                {
                    eldekiAyDagilim[VeriErisim.DegerAl(dr, 0, -1)] = VeriErisim.DegerAl(dr, 1, (double)0.0);
                }
                dr.Close();
            }
            else
                hataStr += "Veri tabanı hatası:" + ve.hata + "<br>";
            for (int i = 0; i < 12; i++)
            {
                if (ayDagilim[i] > 0 && eldekiAyDagilim[i] == 0)
                {
                    int bak1;
                    int bak2;
                    if (i == 0 || i == 3 || i == 6 || i == 9)
                    {
                        bak1 = i + 1;
                        bak2 = i + 2;
                    }
                    else if (i == 1 || i == 4 || i == 7 || i == 10)
                    {
                        bak1 = i - 1;
                        bak2 = i + 1;
                    }
                    else
                    {
                        bak1 = i - 1;
                        bak2 = i - 2;
                    }
                    int bulduk = -1;
                    if (eldekiAyDagilim[bak1] != 0) bulduk = bak1;
                    else if (eldekiAyDagilim[bak2] != 0) bulduk = bak2;

                    if (bulduk == -1)
                        hataStr += (i + 1).ToString() + ". ay için hesaplanan rakam: " + ayDagilim[i].ToString("#.###") + " iken, eski girilen verilerdeki toplamı sıfır olduğundan dağıtılamadı!<br>";
                    else
                    {
                        ayDagilim[bulduk] += ayDagilim[i];
                        ayDagilim[i] = 0;
                    }
                }
            }
            Sonuc s = new Sonuc();
            s.DegerAta(hataStr);
            if (s.islemSonuc)
            {
                ve.BeginTransaction();
                for (int i = 0; i < 12 && s.islemSonuc; i++)
                {
                    if (ayDagilim[i] != 0)
                        if (eldekiAyDagilim[i] != 0)
                            s = tab.AylikDagilimYap(yil, kurumsalYapiKod, asama, kurKod, ekoKod, ekoKodlar, i, eldekiAyDagilim[i], ayDagilim[i]);
                        else
                        {
                            s.islemSonuc = false;
                            s.hataStr = (i + 1).ToString() + ". ay için hesaplanan rakam: " + ayDagilim[i].ToString("#.###") + " iken, eski girilen verilerdeki toplamı sıfır olduğundan dağıtılamadı!<br>";
                        }
                    else if (ayDagilim[i] == 0)
                        s = tab.AylikDagilimSil(yil, kurumsalYapiKod, asama, kurKod, ekoKod, ekoKodlar, i);
                }
                if (s.islemSonuc)
                    ve.Commit();
                else
                    ve.Rollback();
            }
            ve.VeriErisimKapat();
            return s;
        }
